    Interface LightingConfig

    Complete lighting system configuration

    Remarks

    This interface controls all aspects of the lighting system including time of day presets, operational modes, and visual quality settings.

    Example

    const config: LightingConfig = {
      timeOfDay: 'night',
      mode: 'sports',
      quality: 'high',
      floodlightsEnabled: true,
      courtLightsEnabled: true,
      ambientIntensity: 0.5,
      fogDensity: 0.001,
      bloomStrength: 0.7
    };
    interface LightingConfig {
        timeOfDay: TimeOfDay;
        mode: LightingMode;
        quality: AtmosphereQuality;
        floodlightsEnabled: boolean;
        courtLightsEnabled: boolean;
        ambientIntensity: number;
        fogDensity: number;
        bloomStrength: number;
    }
